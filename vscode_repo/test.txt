#!/bin/bash

# Define file names and email details
input_file="yourfile.txt"
temp_file="temp.txt"
result_file="result.txt"
recipient="your-email@example.com"
subject="Processed File Result"
body="Please find the attached processed file."

# Remove the first line
sed '1d' $input_file > $temp_file

# Remove the last 5 lines
head -n -5 $temp_file > $result_file

# Send the email with the result file
echo "$body" | mailx -s "$subject" -a $result_file $recipient
